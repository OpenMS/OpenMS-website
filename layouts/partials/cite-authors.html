{{- /* Show a few authors + always the last; if many remain, show a 10+ chip, else +N. */ -}} {{- $raw := . | string -}} {{- $parts := split $raw ";" -}} {{- $authors := slice -}} {{- range $parts -}} {{- $name := strings.TrimSpace . -}} {{- if $name -}} {{- $authors = $authors | append $name -}} {{- end -}} {{- end -}} {{- $limit := 7 -}} {{- $total := len $authors -}} {{- if le $total $limit -}} {{- delimit $authors "; " -}} {{- else -}} {{- $last := index $authors (sub $total 1) -}} {{- $rest := first (sub $total 1) $authors -}} {{- if le (len $rest) $limit -}} {{- delimit $authors "; " -}} {{- else -}} {{- $visible := first $limit $rest -}} {{- $more := sub (len $rest) $limit -}} {{- $label := printf "+%d" $more -}} {{- if ge $more 10 -}} {{- $label = "10+" -}} {{- end -}} {{- delimit $visible "; " }} {{ $label }}; {{ $last -}} {{- end -}} {{- end -}}
