{{- /* More than 7 authors: a few authors, an ellipsis, then always the last author. */ -}} {{- $raw := . | string -}} {{- $parts := split $raw ";" -}} {{- $authors := slice -}} {{- range $parts -}} {{- $name := strings.TrimSpace . -}} {{- if $name -}} {{- $authors = $authors | append $name -}} {{- end -}} {{- end -}} {{- $limit := 7 -}} {{- $total := len $authors -}} {{- if le $total $limit -}} {{- delimit $authors "; " -}} {{- else -}} {{- /* Up to 7 first, but at least one name hidden (8 authors: 6, "...", the last). */ -}} {{- $last := index $authors (sub $total 1) -}} {{- $rest := first (sub $total 1) $authors -}} {{- $shown := math.Min $limit (sub (len $rest) 1) | int -}} {{- $visible := first $shown $rest -}} {{- $more := sub (len $rest) $shown -}} {{- delimit $visible "; " }} ...; {{ $last -}} {{- end -}}
